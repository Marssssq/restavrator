local UIRefs = {}

-- Безопасное получение встроенного шрифта Unity
local function GetDefaultFont()
    local font = UE.Resources.GetBuiltinResource(typeof(UE.Font), "LegacyRuntime.ttf")
    if font == nil then
        font = UE.Resources.GetBuiltinResource(typeof(UE.Font), "Arial.ttf")
    end
    return font
end

-- Вспомогательная функция создания UI-панели
local function CreatePanel(parent, name, color, anchorMin, anchorMax, pos, size)
    local go = GameObject(name)
    go.transform:SetParent(parent, false)
    local rt = go:AddComponent(typeof(UE.RectTransform))
    rt.anchorMin = anchorMin
    rt.anchorMax = anchorMax
    rt.anchoredPosition = pos
    rt.sizeDelta = size
    local img = go:AddComponent(typeof(UI.Image))
    img.color = color
    return go, rt, img
end

-- Вспомогательная функция создания текста
local function CreateText(parent, name, content, fontSize, color, anchorMin, anchorMax, pos, size, align)
    local go = GameObject(name)
    go.transform:SetParent(parent, false)
    local rt = go:AddComponent(typeof(UE.RectTransform))
    rt.anchorMin = anchorMin
    rt.anchorMax = anchorMax
    rt.anchoredPosition = pos
    rt.sizeDelta = size
    local txt = go:AddComponent(typeof(UI.Text))
    txt.font = GetDefaultFont()
    txt.text = content
    txt.fontSize = fontSize
    txt.color = color
    txt.alignment = align or UE.TextAnchor.MiddleLeft
    txt.horizontalOverflow = UE.HorizontalWrapMode.Wrap
    txt.verticalOverflow = UE.VerticalWrapMode.Overflow
    return txt, rt, go
end

-- Вспомогательная функция создания кнопки
local function CreateButton(parent, name, label, pos, size, bgColor, onClickFunc)
    local go, rt, img = CreatePanel(parent, name, bgColor, Vector2(0.5, 0.5), Vector2(0.5, 0.5), pos, size)
    local btn = go:AddComponent(typeof(UI.Button))
    btn.targetGraphic = img
    local txt = CreateText(go.transform, "Label", label, 15, Color(0.95, 0.92, 0.85, 1), Vector2(0, 0), Vector2(1, 1), Vector2(0, 0), Vector2(-10, -6), UE.TextAnchor.MiddleCenter)
    btn.onClick:AddListener(onClickFunc)
    return go, btn, txt, img
end

-- ПРОЦЕДУРНАЯ ГЕНЕРАЦИЯ 2D-ПРЕДМЕТА И ПОВРЕЖДЕНИЙ
-- ============================================================================
local function GenerateItemTexture(orderId)
    ItemTex = Texture2D(TEX_SIZE, TEX_SIZE, UE.TextureFormat.RGBA32, false)
    ItemTex.filterMode = UE.FilterMode.Point
    DirtMap = {}
    CrackMap = {}
    BaseColors = {}

    local cx, cy = TEX_SIZE / 2, TEX_SIZE / 2

    for y = 0, TEX_SIZE - 1 do
        DirtMap[y] = {}
        CrackMap[y] = {}
        BaseColors[y] = {}
        for x = 0, TEX_SIZE - 1 do
            local dx = x - cx
            local dy = y - cy
            local dist = math.sqrt(dx * dx + dy * dy)
            local col = Color(0.14, 0.11, 0.09, 1) -- Фон сукна стола
            local isItem = false

            if orderId == "watch" then
                if dist < 82 then
                    isItem = true
                    if dist > 70 then
                        col = Color(0.75, 0.72, 0.65, 1) -- Серебряный обод
                    else
                        col = Color(0.92, 0.89, 0.80, 1) -- Эмалевый циферблат
                        -- Стрелки часов
                        if (math.abs(dx) < 2 and dy > 0 and dy < 45) or (math.abs(dy) < 2 and dx > 0 and dx < 32) then
                            col = Color(0.15, 0.15, 0.2, 1)
                        end
                    end
                end
            elseif orderId == "letter" then
                if math.abs(dx) < 75 and math.abs(dy) < 65 then
                    isItem = true
                    col = Color(0.86, 0.80, 0.64, 1) -- Старая бумага
                    -- Строчки текста и штамп
                    if (y % 14 == 0 and math.abs(dx) < 55) then
                        col = Color(0.35, 0.30, 0.45, 1)
                    end
                    if math.abs(dx) < 22 and math.abs(dy + 40) < 12 then
                        col = Color(0.55, 0.25, 0.45, 1) -- Печать полевой почты
                    end
                end
            else
                -- Дагеротип или рамка
                if math.abs(dx) < 72 and math.abs(dy) < 80 then
                    isItem = true
                    if math.abs(dx) > 58 or math.abs(dy) > 66 then
                        col = Color(0.55, 0.42, 0.24, 1) -- Рама
                    else
                        col = Color(0.68, 0.70, 0.74, 1) -- Зеркальная серебряная пластина
                        -- Силуэт портрета в центре
                        if dist < 24 or (math.abs(dx) < 32 and dy < -15 and dy > -55) then
                            col = Color(0.32, 0.33, 0.36, 1)
                        end
                    end
                end
            end

            BaseColors[y][x] = col
            DirtMap[y][x] = 0
            CrackMap[y][x] = 0

            if isItem then
                -- Генерация пятен окислов и грязи
                local noise = math.sin(x * 0.15) * math.cos(y * 0.15)
                if noise > 0.25 then
                    DirtMap[y][x] = 1
                end
                -- Генерация диагональной трещины/надрыва
                if math.abs((x - cx) - (y - cy) * 0.8) < 2.2 and dist < 55 then
                    CrackMap[y][x] = 1
                end
            end
        end
    end
end

-- Перерисовка пикселей предмета с учетом грязи, трещин, пятен и выбранного стиля
local function RefreshItemVisuals()
    local totalDirt = 0
    local cleanedDirt = 0
    local totalCrack = 0
    local fixedCrack = 0

    for y = 0, TEX_SIZE - 1 do
        for x = 0, TEX_SIZE - 1 do
            local col = BaseColors[y][x]

            -- Эффект финального покрытия (Патина vs Как новое)
            if Game.finishProgress > 50 then
                if Game.selectedStyle == "NEW" then
                    col = Color(
                        Mathf.Clamp01(col.r * 1.12),
                        Mathf.Clamp01(col.g * 1.12),
                        Mathf.Clamp01(col.b * 1.15),
                        1
                    )
                elseif Game.selectedStyle == "PATINA" then
                    col = Color(col.r * 0.95, col.g * 0.92, col.b * 0.85, 1)
                end
            end

            -- Отрисовка оставшейся грязи
            if DirtMap[y][x] > 0 then
                totalDirt = totalDirt + 1
                if DirtMap[y][x] == 1 then
                    col = Color(col.r * 0.45, col.g * 0.42, col.b * 0.30, 1)
                else
                    cleanedDirt = cleanedDirt + 1
                end
            end

            -- Отрисовка трещины
            if CrackMap[y][x] > 0 then
                totalCrack = totalCrack + 1
                if CrackMap[y][x] == 1 then
                    col = Color(0.08, 0.05, 0.05, 1)
                else
                    fixedCrack = fixedCrack + 1
                end
            end

            -- Необратимое химическое пятно при ошибке в растворителе или клее
            if Game.chemicalStain then
                local dx = x - (TEX_SIZE * 0.62)
                local dy = y - (TEX_SIZE * 0.58)
                if (dx * dx + dy * dy) < 520 then
                    col = Color(0.45, 0.18, 0.42, 1) -- Едкое фиолетово-бурое пятно
                end
            end

            -- Визуализация стертого слоя (при слишком быстрой чистке мышью)
            if DirtMap[y][x] == -1 then
                col = Color(0.22, 0.20, 0.18, 1) -- Царапина до подложки
            end

            ItemTex:SetPixel(x, y, col)
        end
    end

    ItemTex:Apply()
    UIRefs.itemImage.sprite = Sprite.Create(ItemTex, Rect(0, 0, TEX_SIZE, TEX_SIZE), Vector2(0.5, 0.5))

    if totalDirt > 0 then
        Game.cleanProgress = math.floor((cleanedDirt / totalDirt) * 100)
    else
        Game.cleanProgress = 100
    end

    if totalCrack > 0 then
        Game.glueProgress = math.floor((fixedCrack / totalCrack) * 100)
    else
        Game.glueProgress = 100
    end
end

-- ============================================================================
-- УПРАВЛЕНИЕ ФАЗАМИ ЗАКАЗА
-- ============================================================================
local function GetCurrentOrder()
    if Game.currentOrderIdx == 3 and Game.secretViolated then
        return Orders[4] -- Подмена 3-го заказа из-за потери доверия!
    end
    return Orders[Game.currentOrderIdx]
end

local function UpdateTopBar()
    local order = GetCurrentOrder()
    UIRefs.headerTitle.text = order and order.title or "Итоги смены"
    UIRefs.statsText.text = string.format("Касса: %d руб.   |   Доверие: %d%%   |   Износ слоя: %d%%",
        Game.money, Game.trust, math.floor(Game.overScrubDamage))
end

local function SetPhase(newPhase)
    Game.phase = newPhase
    local order = GetCurrentOrder()

    -- Скрываем всплывающие окна по умолчанию
    UIRefs.modalPanel:SetActive(false)
    UIRefs.materialsPanel:SetActive(false)
    UIRefs.restoreToolbar:SetActive(false)
    UIRefs.loupeObj:SetActive(false)

    if newPhase == "BRIEFING" then
        Game.zonesDiscovered = 0
        Game.cleanProgress = 0
        Game.glueProgress = 0
        Game.finishProgress = 0
        Game.chemicalStain = false
        Game.overScrubDamage = 0
        Game.selectedSolvent = nil
        Game.selectedGlue = nil
        Game.selectedStyle = nil
        Game.twistTriggered = false
        Game.twistResolved = false
        Game.solventEmpty = false

        for _, d in ipairs(order.diagnostics) do d.found = false end
        GenerateItemTexture(order.id)
        RefreshItemVisuals()

        UIRefs.modalPanel:SetActive(true)

-- ============================================================================
-- ОЦЕНКА ЗАКАЗА И ИТОГОВАЯ КАРТОЧКА ИСТОРИИ
-- ============================================================================
function EvaluateOrder()
    local order = GetCurrentOrder()
    local pay = order.baseReward
    local notes = {}

    -- 1. Проверка правильности материалов
    if Game.chemicalStain then
        pay = math.floor(pay * 0.45)
        table.insert(notes, "• НЕВЕРНЫЙ СОСТАВ: На предмете осталось необратимое химическое пятно (-55% ценности).")
    else
        table.insert(notes, "• МАТЕРИАЛЫ: Растворитель и клей подобраны безупречно.")
    end

    -- 2. Проверка стертого слоя от спама мышью
    if Game.overScrubDamage > 25 then
        pay = math.floor(pay * 0.6)
        table.insert(notes, string.format("• СЛИШКОМ РЕЗКАЯ ЧИСТКА: Вы стерли базовый слой на %d%% (-40%% ценности).", math.floor(Game.overScrubDamage)))
    else
        table.insert(notes, "• ТЕМП РАБОТЫ: Бережные и плавные движения сохранили фактуру.")
    end

    -- 3. Оценка выбора «Сохранить следы возраста» vs «Как новое»
    if Game.selectedStyle == order.preferredStyle then
        if order.clientType == "COLLECTOR" then
            pay = pay + 500
            table.insert(notes, "• ВЫБОР СТИЛЯ: Коллекционер восхищен сохраненными следами возраста (+500 руб. премии)!")
        else
            pay = pay + 300
            table.insert(notes, "• ВЫБОР СТИЛЯ: Заказчик счастлив, что вещь выглядит «как новая» (+300 руб.)!")
        end
    else
        pay = math.floor(pay * 0.7)
        if order.clientType == "COLLECTOR" then
            table.insert(notes, "• ВЫБОР СТИЛЯ: Вы сделали предмет «как новый». Коллекционер возмущен потерей исторической патины (-30% оплаты)!")
        else
            table.insert(notes, "• ВЫБОР СТИЛЯ: Вы оставили следы старости, хотя обычный заказчик просил вид «как новое» (-30% оплаты)!")
        end
    end

    Game.money = Game.money + pay

    -- Формирование истории предмета
    local storyText = order.histories[Game.selectedStyle] or order.histories.PATINA
    if order.id == "watch" and Game.secretViolated then
        storyText = order.histories.SECRET_STOLEN
        table.insert(notes, "• ВСКРЫТ ТАЙНИК: Заказчик заметил сорванную внутреннюю пломбу! Доверие упало, элитные клиенты больше не придут.")
    elseif (Game.chemicalStain or Game.overScrubDamage > 40) and order.histories.RUINED then
        storyText = order.histories.RUINED
    end

    table.insert(Game.historyLog, string.format("■ %s\nИтог: %s (Оплата: %d руб.)", order.title, storyText, pay))

    UIRefs.modalPanel:SetActive(true)
    UIRefs.modalTitle.text = "КАРТОЧКА ПРЕДМЕТА: " .. order.title
    UIRefs.modalBody.text = string.format(
        "ОЦЕНКА РАБОТЫ (Гонорар: %d руб.):\n%s\n\nИСТОРИЯ ПРЕДМЕТА:\n«%s»",
        pay, table.concat(notes, "\n"), storyText
    )
    UIRefs.modalBtn1.gameObject:SetActive(true)
    UIRefs.modalBtn1Text.text = (Game.currentOrderIdx < 3) and "Перейти к следующему заказу ->" or "Подвести итоги работы мастерской"
    UIRefs.modalBtn2.gameObject:SetActive(false)
    UpdateTopBar()
end

function ShowFinale()
    UIRefs.modalPanel:SetActive(true)
    local rank = "Мастер-хранитель истории"
    if Game.secretViolated then
        rank = "Любопытный ремесленник (Доверие коллекционеров утрачено)"
    elseif Game.money < 2000 then
        rank = "Начинающий подмастерье"
    end

    UIRefs.modalTitle.text = "СМЕНА ОКОНЧЕНА — ИТОГИ МАСТЕРСКОЙ"
    UIRefs.modalBody.text = string.format(
        "Статус: %s\nЗаработано: %d руб. | Итоговое доверие: %d%%\n\nАРХИВ ВОССТАНОВЛЕННЫХ ИСТОРИЙ:\n%s",
        rank, Game.money, Game.trust, table.concat(Game.historyLog, "\n\n")
    )
    UIRefs.modalBtn1.gameObject:SetActive(true)
    UIRefs.modalBtn1Text.text = "Начать новую смену заново"
    UIRefs.modalBtn2.gameObject:SetActive(false)
end

-- ============================================================================
-- ОБРАБОТКА СЮЖЕТНЫХ ВЫБОРОВ В МОДАЛЬНОМ ОКНЕ
-- ============================================================================
function OnModalChoice(choiceIdx)
    local order = GetCurrentOrder()

    if Game.phase == "BRIEFING" then
        SetPhase("DIAGNOSIS")

    elseif Game.phase == "TWIST" then
        Game.twistResolved = true
        if order.twistType == "SECRET_BOX" then
            if choiceIdx == 1 then
                -- Игрок вскрыл тайник без разрешения!
                Game.secretViolated = true
                Game.trust = math.max(0, Game.trust - 55)
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[ТАЙНИК ВСКРЫТ]: Внутри записка о спрятанном кладе, но сургучная пломба сломана. Доверие к мастерской упало!"
            else
                Game.trust = math.min(100, Game.trust + 10)
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[ТАЙНИК СОХРАНЕН]: Вы аккуратно закрыли крышку, сохранив тайну владельца."
            end
        elseif order.twistType == "OUT_OF_MATERIAL" then
            if choiceIdx == 1 then
                Game.money = Game.money - 100
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[ЗАМЕНА НАЙДЕНА]: Вы использовали чистый конденсат (-100 руб.), бумага спасена."
            else
                Game.chemicalStain = true
                RefreshItemVisuals()
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[ОШИБКА]: Ацетон оставил расплывчатое пятно на фронтовом письме!"
            end
        elseif order.twistType == "CLIENT_CHANGES_MIND" then
            order.preferredStyle = "NEW" -- Заказчик теперь требует «Как новое»!
            if choiceIdx == 1 then
                Game.selectedStyle = "NEW"
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[НОВЫЕ ТРЕБОВАНИЯ]: Режим переключен на восстановление «Как новое»."
            else
                Game.selectedStyle = "PATINA"
                UIRefs.logText.text = UIRefs.logText.text .. "\n\n[ПРИНЦИПИАЛЬНОСТЬ]: Вы решили сохранить историческую патину вопреки звонку."
            end
            RefreshItemVisuals()
        end
        SetPhase("RESTORE")

    elseif Game.phase == "RESULT" then
        if Game.currentOrderIdx < 3 then
            Game.currentOrderIdx = Game.currentOrderIdx + 1
            SetPhase("BRIEFING")
        else
            SetPhase("FINALE")
        end

    elseif Game.phase == "FINALE" then
        Game.currentOrderIdx = 1
        Game.trust = 100
        Game.money = 0
        Game.secretViolated = false
        Game.historyLog = {}
        Orders[3].preferredStyle = "PATINA"
        SetPhase("BRIEFING")
    end
end

-- ============================================================================
-- ПОКАДРОВАЯ ОБРАБОТКА МЫШИ (ЛУПА, ОЧИСТКА, СКЛЕЙКА И ЗАЩИТА ОТ СПАМА)
-- ============================================================================
function Update(dt)
    local mousePos = Input.mousePosition

    -- Вычисление скорости мыши для защиты от спама
    if Game.lastMousePos ~= nil and dt > 0 then
        local dist = Vector3.Distance(mousePos, Game.lastMousePos)
        Game.mouseSpeed = Mathf.Lerp(Game.mouseSpeed, dist / dt, dt * 12)
    end
    Game.lastMousePos = mousePos

    -- Обновление полоски темпа движения мыши
    local speedRatio = Mathf.Clamp01(Game.mouseSpeed / 1400)
    UIRefs.speedFillRT.sizeDelta = Vector2(360 * speedRatio, 0)
    if speedRatio > 0.68 then
        UIRefs.speedFillImg.color = Color(0.9, 0.25, 0.2, 1) -- Опасная скорость!
    else
        UIRefs.speedFillImg.color = Color(0.3, 0.8, 0.4, 1) -- Безопасный темп
    end

    -- Перевод координат мыши в локальные координаты предмета (от -180 до +180)
    local isInside, localPt = RectTransformUtility.ScreenPointToLocalPointInRectangle(UIRefs.itemRT, Vector2(mousePos.x, mousePos.y), nil)

    if Game.phase == "DIAGNOSIS" and isInside then
        UIRefs.loupeRT.anchoredPosition = localPt
        if Input.GetMouseButton(0) then
            local order = GetCurrentOrder()
            for _, d in ipairs(order.diagnostics) do
                if not d.found then
                    local dx = localPt.x - d.x
                    local dy = localPt.y - d.y
                    if math.sqrt(dx * dx + dy * dy) < d.r then
                        d.found = true
                        Game.zonesDiscovered = Game.zonesDiscovered + 1
                        UIRefs.logText.text = UIRefs.logText.text .. "\n\n• " .. d.text
                        UIRefs.hintText.text = string.format("Найдено повреждений: %d / 3", Game.zonesDiscovered)
                        if Game.zonesDiscovered >= 3 then
                            SetPhase("MATERIALS")
                        end
                    end
                end
            end
        end

    elseif Game.phase == "RESTORE" and isInside then
        -- Сводка прогресса слева
        local toolNames = { CLEAN = "Тампон (Очистка)", GLUE = "Шпатель (Склейка)", FINISH = "Кисть (Финиш)" }
        UIRefs.progressText.text = string.format(
            "Активный инструмент:\n[%s]\n\nОчистка: %d%%\nСклейка: %d%%\nФиниш: %d%%",
            toolNames[Game.activeTool], Game.cleanProgress, Game.glueProgress, math.floor(Game.finishProgress)
        )

        if Input.GetMouseButton(0) and math.abs(localPt.x) < 180 and math.abs(localPt.y) < 180 then
            -- Перевод в координаты пикселей текстуры (0..TEX_SIZE-1)
            local px = math.floor(((localPt.x + 180) / 360) * TEX_SIZE)
            local py = math.floor(((localPt.y + 180) / 360) * TEX_SIZE)
            local modified = false

            -- Проверка на спам и слишком высокую скорость движения мыши
            local isTooFast = (speedRatio > 0.68)

            if Game.activeTool == "CLEAN" then
                for y = math.max(0, py - 8), math.min(TEX_SIZE - 1, py + 8) do
                    for x = math.max(0, px - 8), math.min(TEX_SIZE - 1, px + 8) do
                        if (x - px) * (x - px) + (y - py) * (y - py) <= 64 then
                            if isTooFast then
                                -- Защита от спама: слишком быстрое трение стирает базовый слой!
                                DirtMap[y][x] = -1
                                Game.overScrubDamage = math.min(100, Game.overScrubDamage + dt * 6.5)
                                modified = true
                            elseif DirtMap[y][x] == 1 then
                                DirtMap[y][x] = 2 -- Аккуратно очищено
                                modified = true
                            end
                        end
                    end
                end

            elseif Game.activeTool == "GLUE" then
                if Game.cleanProgress < 40 then
                    UIRefs.hintText.text = "НАРУШЕН ПОРЯДОК: Нельзя клеить поверх грязи! Сначала очистите хотя бы 40% предмета!"
                else
                    for y = math.max(0, py - 6), math.min(TEX_SIZE - 1, py + 6) do
                        for x = math.max(0, px - 6), math.min(TEX_SIZE - 1, px + 6) do
                            if CrackMap[y][x] == 1 and not isTooFast then
                                CrackMap[y][x] = 2 -- Трещина склеена
                                modified = true
                            end
                        end
                    end
                end

            elseif Game.activeTool == "FINISH" then
                if Game.cleanProgress < 65 or Game.glueProgress < 65 then
                    UIRefs.hintText.text = "НАРУШЕН ПОРЯДОК: Подкраска и консервация наносятся только после очистки и склейки!"
                elseif not isTooFast then
                    Game.finishProgress = math.min(100, Game.finishProgress + dt * 38)
                    modified = true
                end
            end

            if modified then
                RefreshItemVisuals()
                UpdateTopBar()
            end

            -- Проверка триггера смены условий на середине заказа
            local totalProg = (Game.cleanProgress + Game.glueProgress) / 2
            local order = GetCurrentOrder()
            if totalProg >= 45 and not Game.twistTriggered and order.twistType ~= "NONE" then
                Game.twistTriggered = true
                SetPhase("TWIST")
            end
        end
    end
end

function OnDestroy()
    if ItemTex ~= nil then
        UE.Object.Destroy(ItemTex)
    end
end
